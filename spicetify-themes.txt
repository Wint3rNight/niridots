# Spotify theming. Active: SpicetifyCat, sub-theme "Stray".
#
#   git clone --depth 1 https://github.com/Adrien5902/SpicetifyCat.git \
#       ~/.config/spicetify/Themes/SpicetifyCat
#   # sub-theme PNGs live in themes/<Variant>/ but spicetify only copies from
#   # assets/, so they must be staged there:
#   cp ~/.config/spicetify/Themes/SpicetifyCat/themes/Stray/{background,liked_songs}.png \
#      ~/.config/spicetify/Themes/SpicetifyCat/assets/
#   spicetify config current_theme SpicetifyCat color_scheme Stray \
#                     inject_theme_js 1 replace_colors 1 overwrite_assets 1
#   spicetify backup apply
#
# Variants: Stray (dark, suits the gothic theme), Purple, NyanCat, Summer_Sky.
# To switch, restage that variant's PNGs into assets/ then:
#   spicetify config color_scheme <Variant> && spicetify apply
#
# Our own Cathedral scheme (Sleek layout + gothic palette) is still in
# .config/spicetify/Themes/Cathedral/color.ini if you want to go back.
#
# AFTER EVERY SPOTIFY UPDATE (spotify-launcher auto-updates and reverts the
# patch): spicetify backup apply
